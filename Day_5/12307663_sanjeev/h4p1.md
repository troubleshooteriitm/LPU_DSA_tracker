#count vowels and consonants in a word in Python

''' python

    word = input("Enter a word: ")

    vowels = "aeiouAEIOU"
    vowel_count = 0
    consonant_count = 0

    for ch in word:
        if ch.isalpha():  # Only count letters
            if ch in vowels:
            vowel_count += 1
    else:
        consonant_count += 1

    print("Vowels:", vowel_count)
    print("Consonants:", consonant_count)
